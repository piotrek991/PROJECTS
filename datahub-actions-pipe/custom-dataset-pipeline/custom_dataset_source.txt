"""Custom Dataset Semantic Ingestion Source.

This source:
1. Fetches Dataset entities and their schema metadata from DataHub.
2. Formats a rich semantic text representation (descriptions, column definitions, tags, terms).
3. Chunks the synthesized text while preserving column boundaries.
4. Generates embeddings using the configured or auto-discovered embedding provider.
5. Emits SemanticContent aspects (first-class metadata) to DataHub GMS.
6. Tracks incremental state via:
   - Tier 1: DataHub Server-Side Checkpointing (Stateful Ingestion)
   - Tier 2: Local JSON State File (CLI fallback)
"""

@dataclass
class DatasetSemanticReport(StatefulIngestionReport):
    """Report for dataset semantic ingestion with stateful metrics."""

    num_datasets_fetched: int = 0
    num_datasets_processed: int = 0
    num_datasets_skipped_unchanged: int = 0
    num_datasets_skipped_empty: int = 0
    num_datasets_skipped_filter: int = 0
    num_chunks_created: int = 0
    num_embeddings_generated: int = 0
    num_embedding_failures: int = 0
    processed_datasets: List[str] = field(default_factory=list)
    skipped_datasets: List[str] = field(default_factory=list)
    embedding_failures: List[str] = field(default_factory=list)
    processing_errors: List[str] = field(default_factory=list)

    def report_dataset_fetched(self) -> None:
        self.num_datasets_fetched += 1

    def report_dataset_processed(self, urn: str, num_chunks: int) -> None:
        self.num_datasets_processed += 1
        self.num_chunks_created += num_chunks
        self.processed_datasets.append(f"{urn} ({num_chunks} chunks)")

    def report_dataset_skipped_unchanged(self, urn: Optional[str] = None) -> None:
        self.num_datasets_skipped_unchanged += 1
        if urn:
            self.skipped_datasets.append(f"{urn} (unchanged content hash)")

    def report_dataset_skipped_empty(self, urn: Optional[str] = None) -> None:
        self.num_datasets_skipped_empty += 1
        if urn:
            self.skipped_datasets.append(f"{urn} (empty or text too short)")

    def report_dataset_skipped_filter(self, urn: Optional[str] = None) -> None:
        self.num_datasets_skipped_filter += 1
        if urn:
            self.skipped_datasets.append(f"{urn} (filter mismatch)")

    def report_embeddings_generated(self, count: int) -> None:
        self.num_embeddings_generated += count

    def report_embedding_failure(self, urn: str, error: str) -> None:
        self.num_embedding_failures += 1
        self.embedding_failures.append(f"{urn}: {error}")

    def report_error(self, error: str) -> None:
        self.processing_errors.append(error)


@platform_name("DatasetSemantic", id="dataset-semantic")
@support_status(SupportStatus.INCUBATING)
@config_class(DatasetSemanticSourceConfig)
class DatasetSemanticSource(StatefulIngestionSourceBase):
    """DataHub Source to generate vector embeddings for datasets and schemas."""

    def __init__(self, ctx: PipelineContext, config: DatasetSemanticSourceConfig):
        base_config = cast(StatefulIngestionConfigBase[StatefulIngestionConfig], config)
        super().__init__(base_config, ctx)
        self.config = config
        self.report: DatasetSemanticReport = DatasetSemanticReport()

        self.state_handler: Optional[DatasetChunkingStateHandler] = (
            DatasetChunkingStateHandler(
                source=self,
                config=self.config,
                pipeline_name=self.ctx.pipeline_name,
                run_id=self.ctx.run_id,
            )
            if self.state_provider.is_stateful_ingestion_configured()
            else None
        )

        if ctx.graph:
            self.graph = ctx.graph
        else:
            graph_config = DatahubClientConfig(
                server=config.datahub.server,
                token=config.datahub.token,
                timeout_sec=config.datahub.timeout_seconds,
                disable_ssl_verification=config.datahub.disable_ssl_verification,
            )
            self.graph = DataHubGraph(config=graph_config)

        self.resolved_embedding_config = self._resolve_embedding_config(config.embedding)
        self.embedding_provider = self._init_embedding_provider(self.resolved_embedding_config)

        self.dataset_state: Dict[str, Dict[str, Any]] = {}
        self.state_file_path: Optional[Path] = None
        if self.config.incremental.enabled and (
            not self.state_handler or not self.state_handler.is_checkpointing_enabled()
        ):
            self._initialize_local_state_tracking()

        state_mode = (
            "stateful-checkpointing (GMS)"
            if (self.state_handler and self.state_handler.is_checkpointing_enabled())
            else ("local-file" if self.config.incremental.enabled else "disabled")
        )
        logger.info(
            f"Initialized DatasetSemanticSource using model '{self.resolved_embedding_config.model}' "
            f"(key: '{self.resolved_embedding_config.model_embedding_key}'), state tracking: {state_mode}"
        )

    def _resolve_embedding_config(self, user_cfg: EmbeddingConfig) -> EmbeddingConfig:
        """Fetch server embedding configuration from GMS if not explicitly overridden."""
        cfg = user_cfg.copy()

        if cfg.provider and cfg.model:
            if not cfg.model_embedding_key:
                cfg.model_embedding_key = re.sub(r"[^a-zA-Z0-9_]", "_", cfg.model)
            return cfg

        try:
            query = """
            query getSemanticConfig {
              appConfig {
                semanticSearchConfig {
                  enabled
                  enabledEntities
                  embeddingConfig {
                    provider
                    modelId
                    modelEmbeddingKey
                    awsProviderConfig {
                      region
                    }
                  }
                }
              }
            }
            """
            result = self.graph.execute_graphql(query=query)
            app_config = result.get("appConfig", {}).get("semanticSearchConfig", {})
            if app_config and app_config.get("enabled"):
                server_emb = app_config.get("embeddingConfig", {})
                cfg.provider = cfg.provider or server_emb.get("provider", "openai")
                cfg.model = cfg.model or server_emb.get("modelId", "text-embedding-3-large")
                cfg.model_embedding_key = cfg.model_embedding_key or server_emb.get(
                    "modelEmbeddingKey", "text_embedding_3_large"
                )
                aws_cfg = server_emb.get("awsProviderConfig", {})
                if aws_cfg and aws_cfg.get("region"):
                    cfg.aws_region = cfg.aws_region or aws_cfg["region"]
                logger.info(
                    f"Discovered GMS embedding config: provider={cfg.provider}, model={cfg.model}, key={cfg.model_embedding_key}"
                )
            else:
                cfg.provider = cfg.provider or "openai"
                cfg.model = cfg.model or "sdadas/mmlw-retrieval-roberta-large-v2"
                cfg.model_embedding_key = cfg.model_embedding_key or "text_embedding_3_large"
        except Exception as e:
            logger.warning(f"Could not fetch server semantic config ({e}); using defaults.")
            cfg.provider = cfg.provider or "openai"
            cfg.model = cfg.model or "sdadas/mmlw-retrieval-roberta-large-v2"
            cfg.model_embedding_key = cfg.model_embedding_key or "text_embedding_3_large"

        return cfg

    def _init_embedding_provider(self, cfg: EmbeddingConfig) -> EmbeddingProvider:
        """Instantiate embedding provider directly without relying on unstructured chunking_config."""
        import os
        from datahub.ingestion.source.unstructured.embedding_providers.factory import (
            resolve_local_base_url,
        )

        provider = (cfg.provider or "openai").lower()
        endpoint = (
            cfg.endpoint
            or os.environ.get("OPENAI_EMBEDDING_ENDPOINT")
            or os.environ.get("LOCAL_EMBEDDING_ENDPOINT")
        )
        api_key = (
            cfg.api_key
            or os.environ.get("EMBEDDING_API_KEY")
            or os.environ.get("OPENAI_API_KEY")
            or "local"
        )
        model = cfg.model or "sdadas/mmlw-retrieval-roberta-large-v2"
        timeout = cfg.request_timeout

        if provider in ("openai", "local"):
            from datahub.ingestion.source.unstructured.embedding_providers.openai import (
                OpenAIEmbeddingProvider,
            )

            base_url = resolve_local_base_url(endpoint) if endpoint else None
            return OpenAIEmbeddingProvider(
                model=model,
                api_key=api_key,
                base_url=base_url,
                timeout=timeout,
            )
        elif provider in ("bedrock", "aws-bedrock"):
            from datahub.ingestion.source.unstructured.embedding_providers.bedrock import (
                BedrockEmbeddingProvider,
            )

            return BedrockEmbeddingProvider(
                model=model,
                region=cfg.aws_region or "us-west-2",
                timeout=timeout,
            )
        elif provider == "cohere":
            from datahub.ingestion.source.unstructured.embedding_providers.cohere import (
                CohereEmbeddingProvider,
            )

            return CohereEmbeddingProvider(
                model=model,
                api_key=api_key,
                timeout=timeout,
            )
        else:
            from datahub.ingestion.source.unstructured.embedding_providers.openai import (
                OpenAIEmbeddingProvider,
            )

            base_url = resolve_local_base_url(endpoint) if endpoint else None
            return OpenAIEmbeddingProvider(
                model=model,
                api_key=api_key,
                base_url=base_url,
                timeout=timeout,
            )

    def _initialize_local_state_tracking(self) -> None:
        """Initialize local incremental state cache file."""
        if self.config.incremental.state_file_path:
            self.state_file_path = Path(self.config.incremental.state_file_path)
            self.state_file_path.parent.mkdir(parents=True, exist_ok=True)
        else:
            state_dir = Path.home() / ".datahub" / "dataset_semantic_state"
            state_dir.mkdir(parents=True, exist_ok=True)
            pipeline_name = self.ctx.pipeline_name or "default_dataset_semantic"
            self.state_file_path = state_dir / f"{pipeline_name}.json"

        if self.state_file_path.exists():
            try:
                with open(self.state_file_path, "r", encoding="utf-8") as f:
                    self.dataset_state = json.load(f)
                logger.info(f"Loaded local state with {len(self.dataset_state)} datasets from {self.state_file_path}")
            except Exception as e:
                logger.warning(f"Failed to load local state file: {e}")
                self.dataset_state = {}

    def _save_local_state(self) -> None:
        """Persist local state cache to disk."""
        if self.state_file_path:
            try:
                self.state_file_path.parent.mkdir(parents=True, exist_ok=True)
                with open(self.state_file_path, "w", encoding="utf-8") as f:
                    json.dump(self.dataset_state, f, indent=2)
                logger.debug(f"Saved local state for {len(self.dataset_state)} datasets to {self.state_file_path}")
            except Exception as e:
                logger.warning(f"Failed to save local state file: {e}")

    def _get_processing_config_fingerprint(self) -> Dict[str, Any]:
        """Extract configuration values that affect embedding output artifacts.

        Included in content hash so changes in chunk size or embedding model
        automatically trigger reprocessing.
        """
        return {
            "model": self.resolved_embedding_config.model,
            "provider": self.resolved_embedding_config.provider,
            "max_characters": self.config.chunking.max_characters,
            "max_tokens": getattr(self.config.chunking, "max_tokens", 450),
            "include_header": self.config.chunking.include_header_in_every_chunk,
        }

    def _calculate_dataset_hash(self, text: str) -> str:
        """Deterministic content hash combining text and config fingerprint."""
        hash_input = {
            "content": text,
            "config": self._get_processing_config_fingerprint(),
        }
        hash_str = json.dumps(hash_input, sort_keys=True)
        return hashlib.sha256(hash_str.encode("utf-8")).hexdigest()

    def _should_process_dataset(self, dataset_urn: str, text: str) -> bool:
        """Check whether a dataset has changed since the previous run."""
        if self.config.incremental.force_reprocess:
            return True

        current_hash = self._calculate_dataset_hash(text)

        if self.state_handler and self.state_handler.is_checkpointing_enabled():
            previous_hash = self.state_handler.get_dataset_hash(dataset_urn)
            if previous_hash is not None:
                return previous_hash != current_hash
            return True

        if not self.config.incremental.enabled:
            return True

        if dataset_urn not in self.dataset_state:
            return True

        previous_hash = self.dataset_state[dataset_urn].get("content_hash")
        return previous_hash != current_hash

    def _update_dataset_state(self, dataset_urn: str, text: str) -> None:
        """Record the updated state for a dataset in both active state tiers."""
        content_hash = self._calculate_dataset_hash(text)
        last_processed = datetime.utcnow().isoformat()

        if self.state_handler and self.state_handler.is_checkpointing_enabled():
            self.state_handler.update_dataset_state(
                dataset_urn=dataset_urn,
                content_hash=content_hash,
                last_processed=last_processed,
            )

        if self.config.incremental.enabled:
            self.dataset_state[dataset_urn] = {
                "content_hash": content_hash,
                "last_processed": last_processed,
            }

    def get_workunits_internal(self) -> Iterable[MetadataWorkUnit]:
        """Main execution flow: stream datasets, chunk text, generate embeddings, and yield MCPs."""
        processed_count = 0

        for dataset in self._fetch_datasets_graphql():
            if self.config.max_datasets and processed_count >= self.config.max_datasets:
                logger.info(f"Reached max_datasets limit of {self.config.max_datasets}")
                break

            urn = dataset["urn"]
            self.report.report_dataset_fetched()

            header_text, column_texts, table_anchor = self._create_dataset_text_elements(dataset)
            combined_text = (header_text + " " + " ".join(column_texts)).strip()

            if len(combined_text.strip()) < self.config.min_text_length:
                logger.debug(f"Skipping {urn} (text content too short: {len(combined_text)} chars)")
                self.report.report_dataset_skipped_empty(urn)
                continue

            if not self._should_process_dataset(urn, combined_text):
                logger.debug(f"Skipping {urn} (unchanged content hash)")
                self.report.report_dataset_skipped_unchanged(urn)
                continue

            chunks = self._chunk_dataset_elements(header_text, column_texts, table_anchor)
            if not chunks:
                continue

            try:
                embeddings = self._generate_embeddings(chunks)
            except Exception as e:
                self.report.report_embedding_failure(urn, str(e))
                logger.error(f"Failed to generate embeddings for {urn}: {e}", exc_info=True)
                continue

            yield from self._emit_semantic_content(urn, chunks, embeddings)

            self._update_dataset_state(urn, combined_text)

            processed_count += 1
            self.report.report_dataset_processed(urn, len(chunks))
            logger.info(f"✅ Processed dataset [{processed_count}]: {urn} ({len(chunks)} chunks, {len(embeddings)} vectors)")

        is_dry_run = getattr(self.ctx, "dry_run_mode", False) or getattr(self.ctx, "dry_run", False) or getattr(self.ctx, "preview_mode", False)
        if self.config.incremental.enabled and not is_dry_run:
            self._save_local_state()

    def _fetch_datasets_graphql(self) -> Iterable[Dict[str, Any]]:
        """Stream datasets from DataHub GraphQL API with server-side pagination."""
        patterns = getattr(self.config, "dataset_patterns", None) or getattr(
            self.config, "dataset_filter", None
        )
        
        gql_search_query = "*"
        if patterns and len(patterns) == 1 and patterns[0]:
            raw_p = str(patterns[0]).strip()
            clean_search = raw_p.replace("*", "").replace("?", "").strip()
            if clean_search and len(clean_search) >= 2:
                gql_search_query = clean_search

        logger.info(
            f"Fetching datasets from DataHub GraphQL (query: '{gql_search_query}', "
            f"patterns: {patterns}, platform_filter: {getattr(self.config, 'platform_filter', None)})"
        )

        query = """
        query getDatasetsForSemanticSearch($start: Int!, $count: Int!, $query: String!) {
          searchAcrossEntities(
            input: {
              types: [DATASET]
              query: $query
              start: $start
              count: $count
            }
          ) {
            total
            searchResults {
              entity {
                urn
                type
                ... on Dataset {
                  name
                  platform {
                    name
                  }
                  properties {
                    name
                    description
                    customProperties {
                      key
                      value
                    }
                  }
                  editableProperties {
                    description
                  }
                  schemaMetadata {
                    fields {
                      fieldPath
                      type
                      nativeDataType
                      description
                      tags {
                        tags {
                          tag {
                            name
                          }
                        }
                      }
                      glossaryTerms {
                        terms {
                          term {
                            name
                          }
                        }
                      }
                      schemaFieldEntity {
                        structuredProperties {
                          properties {
                            structuredProperty {
                              definition {
                                displayName
                                qualifiedName
                              }
                            }
                            values {
                              ... on StringValue {
                                stringValue
                              }
                              ... on NumberValue {
                                numberValue
                              }
                            }
                            valueEntities {
                              urn
                              type
                              ... on CorpUser {
                                username
                              }
                              ... on GlossaryTerm {
                                name
                              }
                            }
                          }
                        }
                      }
                    }
                  }
                  tags {
                    tags {
                      tag {
                        name
                      }
                    }
                  }
                  glossaryTerms {
                    terms {
                      term {
                        name
                      }
                    }
                  }
                  structuredProperties {
                    properties {
                      structuredProperty {
                        definition {
                          displayName
                          qualifiedName
                        }
                      }
                      values {
                        ... on StringValue {
                          stringValue
                        }
                        ... on NumberValue {
                          numberValue
                        }
                      }
                      valueEntities {
                        urn
                        type
                        ... on CorpUser {
                          username
                        }
                        ... on GlossaryTerm {
                          name
                        }
                      }
                    }
                  }
                }
              }
            }
          }
        }
        """

        start = 0
        batch_size = 50

        while True:
            try:
                response = self.graph.execute_graphql(
                    query=query,
                    variables={"start": start, "count": batch_size, "query": gql_search_query},
                )
                search_data = response.get("searchAcrossEntities", {})
                total = search_data.get("total", 0)
                results = search_data.get("searchResults", [])

                if not results:
                    break

                for item in results:
                    entity = item.get("entity")
                    if entity:
                        plat_filter = getattr(self.config, "platform_filter", None)
                        if plat_filter:
                            plat = entity.get("platform", {}).get("name", "").lower()
                            if plat not in [p.lower() for p in plat_filter]:
                                continue

                        if not self._matches_dataset_filter(entity):
                            self.report.report_dataset_skipped_filter(entity.get("urn"))
                            continue

                        logger.info(f"Dataset matched filter for processing: {entity.get('urn')}")
                        yield entity

                start += len(results)
                if start >= total:
                    break

            except Exception as e:
                logger.error(f"GraphQL dataset query error: {e}", exc_info=True)
                break

    def _matches_dataset_filter(self, entity: Dict[str, Any]) -> bool:
        """Check if dataset matches configured dataset_patterns or dataset_filter."""
        patterns = getattr(self.config, "dataset_patterns", None) or getattr(
            self.config, "dataset_filter", None
        )
        if not patterns:
            return True

        urn = str(entity.get("urn", "")).strip().lower()
        name = str(entity.get("name", "")).strip().lower()
        props = entity.get("properties") or {}
        prop_name = str(props.get("name", "")).strip().lower()

        for pattern in patterns:
            if not pattern:
                continue
            p = str(pattern).strip().lower()

            if p == "*":
                return True

            if p == urn or p == name or (prop_name and p == prop_name):
                return True

            if fnmatch.fnmatch(urn, p) or fnmatch.fnmatch(name, p) or (prop_name and fnmatch.fnmatch(prop_name, p)):
                return True

            clean_p = p.replace("*", "").replace("?", "").strip()
            if clean_p and (clean_p in urn or clean_p in name or (prop_name and clean_p in prop_name)):
                return True

        return False

    def _extract_structured_properties(self, props_container: Optional[Dict[str, Any]]) -> List[str]:
        """Extract structured property key-value pairs (displayName/qualifiedName: value)."""
        if not props_container or not props_container.get("properties"):
            return []

        result: List[str] = []
        for prop in props_container.get("properties", []):
            prop_def = (prop.get("structuredProperty") or {}).get("definition") or {}
            prop_name = (prop_def.get("displayName") or prop_def.get("qualifiedName") or "").strip()

            vals: List[str] = []
            for v in prop.get("values", []) or []:
                if "stringValue" in v and v["stringValue"]:
                    vals.append(str(v["stringValue"]).strip())
                elif "numberValue" in v and v["numberValue"] is not None:
                    vals.append(str(v["numberValue"]).strip())

            for entity in prop.get("valueEntities", []) or []:
                if entity.get("name"):
                    vals.append(str(entity["name"]).strip())
                elif entity.get("username"):
                    vals.append(str(entity["username"]).strip())

            unique_vals = [v for i, v in enumerate(vals) if v and v not in vals[:i]]
            if unique_vals:
                joined_vals = ", ".join(unique_vals)
                if prop_name:
                    result.append(f"{prop_name}: {joined_vals}")
                else:
                    result.append(joined_vals)

        return result

    def _estimate_token_count(self, text: str) -> int:
        """Accurately estimate token count for BPE/RoBERTa tokenizers (MMLW/E5).
        
        Guarantees safe budget allocation under the 512-token model limit without
        requiring heavy external dependencies.
        """
        if not text:
            return 0
        tokens = re.findall(r"[a-zA-ZąćęłńóśźżĄĆĘŁŃÓŚŹŻ0-9_]+|[^\w\s]", text)
        count = 0
        for tok in tokens:
            if "_" in tok:
                parts = [p for p in tok.split("_") if p]
                count += len(parts) + (len(tok) - len("".join(parts)))
            elif tok.isalnum():
                count += max(1, (len(tok) + 3) // 4)
            else:
                count += 1
        return int(count * 1.1) + 2

    def _is_within_limits(self, text: str) -> bool:
        """Check if text satisfies both character limit and token limit."""
        max_chars = self.config.chunking.max_characters
        max_tokens = getattr(self.config.chunking, "max_tokens", 450)
        if len(text) > max_chars:
            return False
        return self._estimate_token_count(text) <= max_tokens

    def _create_dataset_text_elements(self, dataset: Dict[str, Any]) -> Tuple[str, List[str], str]:
        """Synthesize clean natural language text elements from dataset metadata.
        
        Chunk 0 focuses on pure business context (table name, platform, full business
        description, tags, glossary terms, domain, owners).
        
        Column elements only include valuable columns (those with descriptions, tags,
        terms, or structured properties), eliminating bare undocumented column noise.
        """
        name = dataset.get("name", "")
        platform = dataset.get("platform", {}).get("name", "")

        base_name_anchor = f"Tabela {name} w {platform}." if platform else f"Tabela {name}."

        desc = ""
        editable_props = dataset.get("editableProperties")
        if editable_props and editable_props.get("description"):
            desc = editable_props["description"]
        elif dataset.get("properties") and dataset["properties"].get("description"):
            desc = dataset["properties"]["description"]

        clean_desc = desc.strip()
        if clean_desc:
            if len(clean_desc) <= 250:
                short_desc = clean_desc
            else:
                first_sent = clean_desc.split(".")[0].strip()
                if 20 <= len(first_sent) <= 250:
                    short_desc = f"{first_sent}."
                else:
                    short_desc = clean_desc[:250].rsplit(" ", 1)[0].strip()
            if not short_desc.endswith("."):
                short_desc += "."
            table_anchor = f"{base_name_anchor} {short_desc}"
        else:
            table_anchor = base_name_anchor

        tags: List[str] = []
        if dataset.get("tags") and dataset["tags"].get("tags"):
            tags = [t["tag"]["name"] for t in dataset["tags"]["tags"] if t.get("tag")]

        terms: List[str] = []
        if dataset.get("glossaryTerms") and dataset["glossaryTerms"].get("terms"):
            terms = [t["term"]["name"] for t in dataset["glossaryTerms"]["terms"] if t.get("term")]

        struct_props = self._extract_structured_properties(dataset.get("structuredProperties"))

        overview_parts: List[str] = [base_name_anchor]
        if desc:
            overview_parts.append(desc.strip())
        if tags:
            overview_parts.append(f"Tagi: {', '.join(tags)}.")
        if terms:
            overview_parts.append(f"Pojęcia: {', '.join(terms)}.")
        if struct_props:
            overview_parts.append(". ".join(struct_props) + ".")

        header_text = " ".join(overview_parts).strip()

        column_texts: List[str] = []
        schema_metadata = dataset.get("schemaMetadata")
        fields_list = schema_metadata.get("fields", []) if schema_metadata else []

        for f_info in fields_list:
            field_name = f_info.get("fieldPath", "")
            field_desc = (f_info.get("description") or "").strip()

            field_tags: List[str] = []
            if f_info.get("tags") and f_info["tags"].get("tags"):
                field_tags = [t["tag"]["name"] for t in f_info["tags"]["tags"] if t.get("tag")]

            field_terms: List[str] = []
            if f_info.get("glossaryTerms") and f_info["glossaryTerms"].get("terms"):
                field_terms = [t["term"]["name"] for t in f_info["glossaryTerms"]["terms"] if t.get("term")]

            col_struct_props: List[str] = []
            if f_info.get("schemaFieldEntity") and f_info["schemaFieldEntity"].get("structuredProperties"):
                col_struct_props = self._extract_structured_properties(f_info["schemaFieldEntity"]["structuredProperties"])

            if not (field_desc or field_tags or field_terms or col_struct_props):
                continue

            col_parts: List[str] = []
            clean_field_name = field_name.replace("_", " ") if "_" in field_name else field_name
            if field_desc:
                col_parts.append(f"{clean_field_name}: {field_desc}.")
            else:
                col_parts.append(f"{clean_field_name}.")

            if field_tags:
                col_parts.append(f"Tagi: {', '.join(field_tags)}.")
            if field_terms:
                col_parts.append(f"Pojęcia: {', '.join(field_terms)}.")
            if col_struct_props:
                col_parts.append(". ".join(col_struct_props) + ".")

            column_texts.append(" ".join(col_parts))

        return header_text, column_texts, table_anchor

    def _chunk_dataset_elements(
        self, header_text: str, column_texts: List[str], table_anchor: str
    ) -> List[Dict[str, Any]]:
        """Group header and column blocks into token-validated, natural language chunks."""
        max_chars = self.config.chunking.max_characters
        max_tokens = getattr(self.config.chunking, "max_tokens", 450)
        chunks: List[Dict[str, Any]] = []

        if column_texts:
            full_combined = f"{header_text} {' '.join(column_texts)}".strip()
            if self._is_within_limits(full_combined):
                chunks.append({
                    "text": full_combined,
                    "character_offset": 0,
                    "character_length": len(full_combined),
                })
                return chunks

        if self._is_within_limits(header_text):
            chunks.append({
                "text": header_text,
                "character_offset": 0,
                "character_length": len(header_text),
            })
        else:
            for offset in range(0, len(header_text), max_chars):
                sub = header_text[offset : offset + max_chars].strip()
                if sub:
                    chunks.append({
                        "text": sub,
                        "character_offset": offset,
                        "character_length": len(sub),
                    })

        max_chunks = getattr(self.config.chunking, "max_chunks_per_dataset", None)
        if not column_texts:
            return chunks[:max_chunks] if max_chunks else chunks

        anchor_prefix = table_anchor if self.config.chunking.include_header_in_every_chunk else ""
        current_block: List[str] = [anchor_prefix] if anchor_prefix else []
        chunk_offset = sum(len(c["text"]) for c in chunks)

        for col in column_texts:
            candidate_text = " ".join(current_block + [col]).strip()
            
            if not self._is_within_limits(candidate_text) and len(current_block) > (1 if anchor_prefix else 0):
                full_chunk_text = " ".join(current_block).strip()
                chunks.append({
                    "text": full_chunk_text,
                    "character_offset": chunk_offset,
                    "character_length": len(full_chunk_text),
                })
                chunk_offset += len(full_chunk_text)
                current_block = [anchor_prefix] if anchor_prefix else []

            current_block.append(col)

        if len(current_block) > (1 if anchor_prefix else 0):
            full_chunk_text = " ".join(current_block).strip()
            chunks.append({
                "text": full_chunk_text,
                "character_offset": chunk_offset,
                "character_length": len(full_chunk_text),
            })

        if max_chunks and len(chunks) > max_chunks:
            logger.info(f"Capping dataset chunks from {len(chunks)} to {max_chunks}")
            chunks = chunks[:max_chunks]

        return chunks

    def _generate_embeddings(self, chunks: List[Dict[str, Any]]) -> List[List[float]]:
        """Call the embedding provider in batches."""
        texts = [chunk["text"] for chunk in chunks]
        batch_size = self.resolved_embedding_config.batch_size
        all_embeddings: List[List[float]] = []

        for i in range(0, len(texts), batch_size):
            batch = texts[i : i + batch_size]
            result = self.embedding_provider.embed(batch)
            all_embeddings.extend(result.embeddings)

        self.report.report_embeddings_generated(len(all_embeddings))
        return all_embeddings

    def _emit_semantic_content(
        self, urn: str, chunks: List[Dict[str, Any]], embeddings: List[List[float]]
    ) -> Iterable[MetadataWorkUnit]:
        """Construct SemanticContent aspect and yield MetadataWorkUnit."""
        model_version = derive_model_id(
            self.resolved_embedding_config.provider, self.resolved_embedding_config.model
        ) or f"{self.resolved_embedding_config.provider}/{self.resolved_embedding_config.model}"

        embedding_chunks: List[EmbeddingChunkClass] = []
        for i, (chunk, vector) in enumerate(zip(chunks, embeddings)):
            rounded_vector = [round(float(v), 5) for v in vector]
            embedding_chunks.append(
                EmbeddingChunkClass(
                    position=i,
                    vector=rounded_vector,
                    characterOffset=chunk.get("character_offset", 0),
                    characterLength=chunk.get("character_length", len(chunk["text"])),
                    text=chunk.get("text"),
                )
            )

        model_key = self.resolved_embedding_config.model_embedding_key
        model_data = EmbeddingModelDataClass(
            modelVersion=model_version,
            generatedAt=int(datetime.utcnow().timestamp() * 1000),
            chunkingStrategy="dataset_schema_and_metadata_v1",
            totalChunks=len(embedding_chunks),
            chunks=embedding_chunks,
        )

        semantic_content = SemanticContentClass(embeddings={model_key: model_data})
        mcp = MetadataChangeProposalWrapper(entityUrn=urn, aspect=semantic_content)
        workunit = MetadataWorkUnit(id=f"{urn}-semanticContent", mcp=mcp)

        logger.debug(f"Emitted SemanticContent for {urn} with {len(embedding_chunks)} chunks")
        yield workunit

    def get_report(self) -> SourceReport:
        return self.report

    def close(self) -> None:
        is_dry_run = getattr(self.ctx, "dry_run_mode", False) or getattr(self.ctx, "dry_run", False) or getattr(self.ctx, "preview_mode", False)
        if self.config.incremental.enabled and not is_dry_run:
            self._save_local_state()
        super().close()
